No recommendations yet.
Set up your style preferences →

`; return; } recs.forEach((rec, idx) => { const card = document.createElement('div'); card.className = 'rec-card'; const styleKey = (rec.styles && rec.styles.length > 0) ? rec.styles[0] : 'style'; card.dataset.style = styleKey; card.innerHTML = `
${styleKey}
${rec.matchPercentage ? `Match ${rec.matchPercentage}%` : 'Match not scored'}

${rec.title}

${rec.description}

`; grid.appendChild(card); }); } // ── Filter logic ── function applyFilter(filter) { activeFilter = filter; document.querySelectorAll('.filter-pill').forEach(p => { p.classList.toggle('active', p.dataset.filter === filter); }); const filteredRecs = filter === 'all' ? allRecs : allRecs.filter(rec => { const styleKey = (rec.styles && rec.styles.length > 0) ? rec.styles[0] : 'casual'; return styleKey === filter; }); renderCards(filteredRecs); } // ── Load recommendations ── async function loadRecs() { try { const authRes = await fetch('/api/auth/status'); const authData = await authRes.json(); if (!authData.authenticated) { window.location.href = '/login'; return; } const res = await fetch('/api/recommendations'); const data = await res.json(); allRecs = data.recommendations || []; // Build filter pills from unique styles const styleKeys = [...new Set(allRecs.map(rec => (rec.styles && rec.styles.length > 0) ? rec.styles[0] : 'casual'))]; const filterBar = document.getElementById('filterBar'); styleKeys.forEach(s => { const btn = document.createElement('button'); btn.className = 'filter-pill'; btn.dataset.filter = s; btn.textContent = s.charAt(0).toUpperCase() + s.slice(1); btn.addEventListener('click', () => applyFilter(s)); filterBar.appendChild(btn); }); renderCards(allRecs); } catch (err) { console.error(err); document.getElementById('recsGrid').innerHTML = '

Failed to load recommendations.

'; } } // ── Filter pill click handler (All) ── document.querySelector('.filter-pill[data-filter="all"]').addEventListener('click', () => applyFilter('all')); // ── Logout ── document.getElementById('logoutBtn')?.addEventListener('click', async () => { await fetch('/api/auth/logout', { method: 'POST' }); window.location.href = '/login'; }); document.getElementById('generateRecsBtn')?.addEventListener('click', loadRecs);
